# Golden Set Editor — Unified Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the two-mode Golden Set editor (manual vs review) with a single unified master-detail editor where all queries — regardless of origin — share identical editing capabilities, and auto-generation becomes a dialog rather than a separate page.

**Architecture:** Six sequential tasks: backend schema/service changes first, then frontend components (QueryList → QueryEditor → AutoGenerateDialog → GoldenSetEditorPage), ending with deletion of retired components and the generate route. Each task is independently committable and leaves the app in a working state.

**Tech Stack:** Python 3.12 / FastAPI / Pydantic v2 (backend); React 18 / TypeScript / Vite / shadcn/ui / Tailwind CSS (frontend). Tests: pytest (backend), vitest + React Testing Library (frontend).

```python
def _build_generation_prompt(
    self,
    page_text: str,
    page_range: list[int],
    doc_title: str,
    queries_count: int,
    question_types: list[str],
    system_prompt: str | None = None,
) -> list[dict]:
    type_desc = ", ".join(question_types)
    page_desc = ", ".join(str(p) for p in page_range)

    system_msg = system_prompt or DEFAULT_GENERATION_SYSTEM_PROMPT

    user_msg = (
        f"Document: {doc_title}\n"
        f"Pages: {page_desc}\n"
        f"Generate {queries_count} queries of type(s): {type_desc}\n\n"
        f"--- Document Content ---\n{page_text[:50000]}"
    )

    return [
        {"role": "system", "content": system_msg},
        {"role": "user", "content": user_msg},
    ]
```

Replace the entire contents of `frontend/src/components/evaluation/QueryList.tsx`:

```typescript
import { useState, useMemo, useRef, useEffect } from 'react'
import { Plus, Search, X, Check } from 'lucide-react'
import { Button } from '@/components/ui/button'
import { Input } from '@/components/ui/input'
import { ScrollArea } from '@/components/ui/scroll-area'
import { cn } from '@/lib/utils'
import type { GoldenSetQuery, SourceMethod } from '@/types/golden-set'

interface QueryListProps {
  queries: GoldenSetQuery[]
  selectedId: string | null
  onSelect: (id: string) => void
  onAdd: (queryText: string) => void
}

type FilterKey = SourceMethod | 'all'

const FILTER_CHIPS: { key: FilterKey; label: string }[] = [
  { key: 'all', label: 'All' },
  { key: 'manual', label: 'Manual' },
  { key: 'auto_generated', label: 'Auto' },
  { key: 'imported', label: 'Imported' },
]

const ORIGIN_LABEL: Record<SourceMethod, string> = {
  manual: 'Manual',
  auto_generated: 'Auto',
  imported: 'Imported',
}

export function QueryList({ queries, selectedId, onSelect, onAdd }: QueryListProps) {
  const [search, setSearch] = useState('')
  const [filter, setFilter] = useState<FilterKey>('all')
  const [adding, setAdding] = useState(false)
  const [newText, setNewText] = useState('')
  const newInputRef = useRef<HTMLInputElement>(null)

  useEffect(() => {
    if (adding) newInputRef.current?.focus()
  }, [adding])

  const filtered = useMemo(() => {
    return queries.filter((q) => {
      const matchesSearch = search === '' ||
        q.queryText.toLowerCase().includes(search.toLowerCase())
      const matchesFilter = filter === 'all' || q.sourceMethod === filter
      return matchesSearch && matchesFilter
    })
  }, [queries, search, filter])

  const handleAddSubmit = () => {
    const trimmed = newText.trim()
    if (!trimmed) {
      setAdding(false)
      return
    }
    onAdd(trimmed)
    setNewText('')
    setAdding(false)
  }

  return (
    <div className="flex flex-col h-full border rounded-lg">
      {/* Header */}
      <div className="flex items-center justify-between p-3 border-b shrink-0">
        <h3 className="text-sm font-medium">Queries ({queries.length})</h3>
        <Button variant="ghost" size="icon" onClick={() => setAdding(true)}>
          <Plus className="h-4 w-4" />
        </Button>
      </div>

      {/* Search */}
      <div className="px-3 pt-2 pb-1 shrink-0">
        <div className="relative">
          <Search className="absolute left-2.5 top-2.5 h-3.5 w-3.5 text-muted-foreground" />
          <Input
            className="pl-8 h-8 text-sm"
            placeholder="Search queries…"
            value={search}
            onChange={(e) => setSearch(e.target.value)}
          />
          {search && (
            <button
              className="absolute right-2.5 top-2.5 text-muted-foreground hover:text-foreground"
              onClick={() => setSearch('')}
            >
              <X className="h-3.5 w-3.5" />
            </button>
          )}
        </div>
      </div>

      {/* Filter chips */}
      <div className="px-3 pb-2 flex gap-1.5 flex-wrap shrink-0">
        {FILTER_CHIPS.map(({ key, label }) => (
          <button
            key={key}
            onClick={() => setFilter(key)}
            className={cn(
              'text-xs px-2 py-0.5 rounded-full border transition-colors',
              filter === key
                ? 'bg-primary text-primary-foreground border-primary'
                : 'border-border text-muted-foreground hover:border-foreground hover:text-foreground'
            )}
          >
            {label}
          </button>
        ))}
      </div>

      {/* List */}
      <ScrollArea className="flex-1">
        <div className="p-1">
          {filtered.length === 0 && !adding ? (
            <p className="text-sm text-muted-foreground text-center py-8 px-4">
              {queries.length === 0
                ? 'No queries yet. Click + to add one.'
                : 'No queries match this filter.'}
            </p>
          ) : (
            filtered.map((q) => (
              <button
                key={q.id}
                onClick={() => onSelect(q.id)}
                className={cn(
                  'w-full text-left px-3 py-2 rounded-md text-sm transition-colors',
                  'hover:bg-accent',
                  selectedId === q.id && 'bg-accent border-l-2 border-l-primary'
                )}
              >
                <p className="whitespace-pre-wrap break-words">{q.queryText}</p>
                <p className="text-xs text-muted-foreground mt-0.5">
                  {q.sources.length} source{q.sources.length !== 1 ? 's' : ''}
                  {' · '}
                  <span className="opacity-70">{ORIGIN_LABEL[q.sourceMethod]}</span>
                </p>
              </button>
            ))
          )}

          {/* Inline add */}
          {adding && (
            <div className="px-3 py-2 space-y-1.5">
              <Input
                ref={newInputRef}
                value={newText}
                onChange={(e) => setNewText(e.target.value)}
                placeholder="Enter query text…"
                className="text-sm"
                onKeyDown={(e) => {
                  if (e.key === 'Enter' && !e.shiftKey) {
                    e.preventDefault()
                    handleAddSubmit()
                  }
                  if (e.key === 'Escape') {
                    setAdding(false)
                    setNewText('')
                  }
                }}
              />
              <div className="flex gap-1">
                <Button size="sm" variant="ghost" onClick={handleAddSubmit} disabled={!newText.trim()}>
                  <Check className="h-4 w-4" />
                </Button>
                <Button size="sm" variant="ghost" onClick={() => { setAdding(false); setNewText('') }}>
                  <X className="h-4 w-4" />
                </Button>
              </div>
            </div>
          )}
        </div>
      </ScrollArea>
    </div>
  )
}
```

- [ ] **Step 5.3: Create AutoGenerateDialog**

Create `frontend/src/components/evaluation/AutoGenerateDialog.tsx`:

```typescript
import { useState, useEffect, useRef } from 'react'
import {
  Dialog,
  DialogContent,
  DialogHeader,
  DialogTitle,
  DialogFooter,
} from '@/components/ui/dialog'
import { Button } from '@/components/ui/button'
import { Label } from '@/components/ui/label'
import { Input } from '@/components/ui/input'
import { Checkbox } from '@/components/ui/checkbox'
import { Textarea } from '@/components/ui/textarea'
import {
  Select,
  SelectContent,
  SelectItem,
  SelectTrigger,
  SelectValue,
} from '@/components/ui/select'
import { PromptConfigEditor } from '@/components/shared/PromptConfigEditor'
import { useDocuments } from '@/hooks/useDocuments'
import { DEFAULT_GENERATION_SYSTEM_PROMPT } from '@/constants/golden-set-generation'
import * as api from '@/api/golden-sets'
import type { PromptConfig } from '@/types/prompt-config'
import type { QuestionType } from '@/types/golden-set'

type DialogStep = 'config' | 'progress' | 'complete'

interface AutoGenerateDialogProps {
  open: boolean
  onOpenChange: (open: boolean) => void
  projectId: string
  goldenSetId: string
  initialQueryCount: number
  onComplete: () => void
}

const QUESTION_TYPES: { value: QuestionType; label: string }[] = [
  { value: 'factual', label: 'Factual' },
  { value: 'comparison', label: 'Comparison' },
  { value: 'summarization', label: 'Summarization' },
]

export function AutoGenerateDialog({
  open,
  onOpenChange,
  projectId,
  goldenSetId,
  initialQueryCount,
  onComplete,
}: AutoGenerateDialogProps) {
  const { documents } = useDocuments(projectId, 'ready')

  const [step, setStep] = useState<DialogStep>('config')
  const [documentIds, setDocumentIds] = useState<string[]>([])
  const [queriesPerDocument, setQueriesPerDocument] = useState(5)
  const [questionTypes, setQuestionTypes] = useState<QuestionType[]>(['factual'])
  const [promptConfig, setPromptConfig] = useState<PromptConfig>({
    provider: 'openai',
    model: 'gpt-4o',
    temperature: 0.7,
    systemPrompt: DEFAULT_GENERATION_SYSTEM_PROMPT,
  })
  const [progress, setProgress] = useState<{ completed: number; total: number } | null>(null)
  const [generatedCount, setGeneratedCount] = useState(0)
  const [error, setError] = useState<string | null>(null)
  const pollRef = useRef<ReturnType<typeof setInterval> | null>(null)

  // Reset when dialog opens
  useEffect(() => {
    if (open) {
      setStep('config')
      setDocumentIds([])
      setQueriesPerDocument(5)
      setQuestionTypes(['factual'])
      setPromptConfig({
        provider: 'openai',
        model: 'gpt-4o',
        temperature: 0.7,
        systemPrompt: DEFAULT_GENERATION_SYSTEM_PROMPT,
      })
      setProgress(null)
      setGeneratedCount(0)
      setError(null)
    }
    return () => {
      if (pollRef.current) clearInterval(pollRef.current)
    }
  }, [open])

  const canGenerate = documentIds.length > 0 && questionTypes.length > 0

  const toggleDocument = (id: string) => {
    setDocumentIds((prev) =>
      prev.includes(id) ? prev.filter((d) => d !== id) : [...prev, id]
    )
  }

  const toggleQuestionType = (type: QuestionType) => {
    setQuestionTypes((prev) =>
      prev.includes(type) ? prev.filter((t) => t !== type) : [...prev, type]
    )
  }

  const handleGenerate = async () => {
    setStep('progress')
    setProgress({ completed: 0, total: 0 })

    try {
      await api.generateGoldenSet(projectId, goldenSetId, {
        documentIds,
        llmProvider: promptConfig.provider ?? 'openai',
        llmModel: promptConfig.model ?? 'gpt-4o',
        queriesPerDocument,
        questionTypes,
        temperature: promptConfig.temperature ?? 0.7,
        systemPrompt: promptConfig.systemPrompt || undefined,
      })

      // Poll for completion
      pollRef.current = setInterval(async () => {
        try {
          const gs = await api.getGoldenSet(projectId, goldenSetId)
          const prog = gs.generationProgress
          if (prog) {
            setProgress({ completed: prog.completedWindows, total: prog.totalWindows })
          }

          if (gs.generationStatus === 'completed') {
            clearInterval(pollRef.current!)
            const newCount = Math.max(0, gs.queries.length - initialQueryCount)
            setGeneratedCount(newCount)
            setStep('complete')
            onComplete()
          } else if (gs.generationStatus === 'failed') {
            clearInterval(pollRef.current!)
            const newCount = Math.max(0, gs.queries.length - initialQueryCount)
            setGeneratedCount(newCount)
            setError(prog?.errorMessage ?? 'Generation failed.')
            setStep('complete')
            onComplete()
          }
        } catch {
          // ignore transient poll errors
        }
      }, 2500)
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Failed to start generation.')
      setStep('complete')
    }
  }

  const handleDone = () => {
    onOpenChange(false)
  }

  return (
    <Dialog
      open={open}
      onOpenChange={(v) => {
        // Prevent closing during progress
        if (step === 'progress') return
        onOpenChange(v)
      }}
    >
      <DialogContent className="max-w-lg max-h-[90vh] overflow-y-auto">
        <DialogHeader>
          <DialogTitle>Auto-Generate Queries</DialogTitle>
        </DialogHeader>

        {step === 'config' && (
          <div className="space-y-5">
            {/* Documents */}
            <div className="space-y-2">
              <Label>Documents</Label>
              {documents.length === 0 ? (
                <p className="text-sm text-muted-foreground">No ready documents found.</p>
              ) : (
                <div className="space-y-1.5 max-h-40 overflow-y-auto border rounded-md p-2">
                  {documents.map((doc) => (
                    <div key={doc.id} className="flex items-center gap-2">
                      <Checkbox
                        id={`doc-${doc.id}`}
                        checked={documentIds.includes(doc.id)}
                        onCheckedChange={() => toggleDocument(doc.id)}
                      />
                      <label htmlFor={`doc-${doc.id}`} className="text-sm cursor-pointer">
                        {doc.title}
                      </label>
                    </div>
                  ))}
                </div>
              )}
            </div>

            {/* Queries per document */}
            <div className="space-y-1.5">
              <Label htmlFor="qpd">Queries per document</Label>
              <Input
                id="qpd"
                type="number"
                min={1}
                max={30}
                value={queriesPerDocument}
                onChange={(e) => setQueriesPerDocument(Math.max(1, Math.min(30, Number(e.target.value))))}
                className="w-24"
              />
            </div>

            {/* Question types */}
            <div className="space-y-1.5">
              <Label>Question types</Label>
              <div className="flex gap-4">
                {QUESTION_TYPES.map(({ value, label }) => (
                  <div key={value} className="flex items-center gap-2">
                    <Checkbox
                      id={`qt-${value}`}
                      checked={questionTypes.includes(value)}
                      onCheckedChange={() => toggleQuestionType(value)}
                    />
                    <label htmlFor={`qt-${value}`} className="text-sm cursor-pointer">{label}</label>
                  </div>
                ))}
              </div>
            </div>

            {/* LLM Configuration */}
            <div className="space-y-2">
              <Label>LLM Configuration</Label>
              <PromptConfigEditor
                value={promptConfig}
                onChange={setPromptConfig}
              />
              <button
                type="button"
                className="text-xs text-muted-foreground underline underline-offset-2 hover:text-foreground"
                onClick={() => setPromptConfig((p) => ({ ...p, systemPrompt: DEFAULT_GENERATION_SYSTEM_PROMPT }))}
              >
                Reset to default prompt
              </button>
            </div>

            <DialogFooter>
              <Button variant="outline" onClick={() => onOpenChange(false)}>
                Cancel
              </Button>
              <Button onClick={handleGenerate} disabled={!canGenerate}>
                Generate
              </Button>
            </DialogFooter>
          </div>
        )}

        {step === 'progress' && (
          <div className="space-y-4 py-4">
            <p className="text-sm text-center text-muted-foreground">
              {progress
                ? `${progress.completed} / ${progress.total || '?'} windows processed`
                : 'Starting generation…'}
            </p>
            <div className="h-2 rounded-full bg-muted overflow-hidden">
              <div
                className="h-full rounded-full bg-primary transition-all duration-500"
                style={{
                  width: progress && progress.total > 0
                    ? `${Math.round((progress.completed / progress.total) * 100)}%`
                    : '0%',
                }}
              />
            </div>
            <p className="text-xs text-center text-muted-foreground">
              This may take a minute. Please wait.
            </p>
          </div>
        )}

        {step === 'complete' && (
          <div className="space-y-4 py-4 text-center">
            {error ? (
              <>
                <p className="text-sm text-destructive">{error}</p>
                {generatedCount > 0 && (
                  <p className="text-sm text-muted-foreground">
                    {generatedCount} {generatedCount === 1 ? 'query was' : 'queries were'} created before the failure.
                  </p>
                )}
              </>
            ) : (
              <p className="text-sm font-medium">
                Generated {generatedCount} {generatedCount === 1 ? 'query' : 'queries'}.
              </p>
            )}
            <DialogFooter className="justify-center">
              <Button onClick={handleDone}>Done</Button>
            </DialogFooter>
          </div>
        )}
      </DialogContent>
    </Dialog>
  )
}
```
